cmake_minimum_required(VERSION 3.20)
project(Threadlight LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
option(THREADLIGHT_DEVELOPER_QA "Enable private QA diagnostics" OFF)
option(THREADLIGHT_BUILD_GAME "Build the raylib game" ON)
option(THREADLIGHT_BUILD_BENCHMARKS "Build standalone CPU microbenchmarks" OFF)
option(THREADLIGHT_USE_PINNED_RAYLIB "Use an explicitly supplied MSVC raylib installation" OFF)
set(THREADLIGHT_RAYLIB_ROOT "" CACHE PATH "Root of the pinned MSVC raylib installation")

if(THREADLIGHT_BUILD_GAME)
  if(THREADLIGHT_USE_PINNED_RAYLIB)
    if(NOT THREADLIGHT_RAYLIB_ROOT OR
       NOT EXISTS "${THREADLIGHT_RAYLIB_ROOT}/include/raylib.h" OR
       NOT EXISTS "${THREADLIGHT_RAYLIB_ROOT}/lib/raylib.lib" OR
       NOT EXISTS "${THREADLIGHT_RAYLIB_ROOT}/lib/cmake/raylib/raylib-config.cmake")
      message(FATAL_ERROR "Set THREADLIGHT_RAYLIB_ROOT to a compatible MSVC raylib installation containing include/raylib.h, lib/raylib.lib and lib/cmake/raylib/raylib-config.cmake.")
    endif()
    set(raylib_DIR "${THREADLIGHT_RAYLIB_ROOT}/lib/cmake/raylib" CACHE PATH "raylib package" FORCE)
    set(raylib_LIBRARY "${THREADLIGHT_RAYLIB_ROOT}/lib/raylib.lib" CACHE FILEPATH "raylib library" FORCE)
    set(raylib_INCLUDE_DIR "${THREADLIGHT_RAYLIB_ROOT}/include" CACHE PATH "raylib headers" FORCE)
  endif()
  find_package(raylib 5.5 CONFIG REQUIRED)
  if(MSVC)
    set(_threadlight_raylib_paths "${raylib_LIBRARY}")
    foreach(_target raylib::raylib raylib)
      if(TARGET ${_target})
        foreach(_property IMPORTED_LOCATION IMPORTED_IMPLIB IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB_RELEASE IMPORTED_LOCATION_DEBUG IMPORTED_IMPLIB_DEBUG)
          get_target_property(_path ${_target} ${_property})
          if(_path AND NOT _path MATCHES "-NOTFOUND$")
            list(APPEND _threadlight_raylib_paths "${_path}")
          endif()
        endforeach()
      endif()
    endforeach()
    foreach(_path IN LISTS _threadlight_raylib_paths)
      if(_path MATCHES "\\.a$")
        message(FATAL_ERROR "MSVC cannot use the selected MinGW raylib archive '${_path}'. Select a matching MSVC raylib package, or use the windows-msvc preset.")
      endif()
    endforeach()
  endif()
  if(WIN32)
    # Keep the game a native GUI application while forwarding the original
    # command line to main(), which owns the diagnostic and rendering checks.
    add_executable(Threadlight WIN32 src/windows_entry.cpp src/main.cpp src/performance_metrics.cpp)
  else()
    add_executable(Threadlight src/main.cpp src/performance_metrics.cpp)
  endif()
  if(THREADLIGHT_DEVELOPER_QA)
    target_compile_definitions(Threadlight PRIVATE THREADLIGHT_DEVELOPER_QA=1)
  endif()
  # Prefer the actual library path when available. Some pkg-config based
  # raylib config packages expose Unix -L/-l flags as one imported link item,
  # which MSVC cannot interpret.
  if(TARGET raylib::raylib)
    target_link_libraries(Threadlight PRIVATE raylib::raylib)
  elseif(raylib_LIBRARY)
    target_link_libraries(Threadlight PRIVATE "${raylib_LIBRARY}")
    target_include_directories(Threadlight PRIVATE ${raylib_INCLUDE_DIRS})
  else()
    target_link_libraries(Threadlight PRIVATE raylib)
  endif()
  if(WIN32)
    target_link_libraries(Threadlight PRIVATE winmm psapi shell32)
  endif()
  if(MSVC)
    # The pinned static raylib package is built with the static CRT (/MT).
    # Match it so the Release executable has one consistent runtime and no
    # LNK4098 LIBCMT/MSVCRT conflict.
    set_property(TARGET Threadlight PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(Threadlight PRIVATE /utf-8)
  endif()
endif()
